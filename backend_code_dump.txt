const { auth } = require('../services/firebaseAdmin')

/**
 * Verify Firebase JWT token from Authorization header
 */
async function verifyToken(req, res, next) {
  try {
    const token = req.headers.authorization?.split('Bearer ')[1]

    if (!token) {
      return res.status(401).json({ error: 'No token provided' })
    }

    const decodedToken = await auth.verifyIdToken(token)
    req.user = decodedToken
    next()
  } catch (error) {
    res.status(401).json({ error: 'Invalid token' })
  }
}

/**
 * Check if user has one of the required roles (use AFTER verifyToken)
 * Usage: requireRole('admin') or requireRole('admin', 'chef')
 */
function requireRole(...roles) {
  return (req, res, next) => {
    if (!req.user || !roles.includes(req.user.role)) {
      return res.status(403).json({ error: 'Insufficient permissions' })
    }
    next()
  }
}

function requirePlatformAdmin(req, res, next) {
  const isPlatformAdmin =
    req.user?.platformAdmin === true ||
    req.user?.superAdmin === true ||
    req.user?.role === 'superadmin'

  if (!isPlatformAdmin) {
    return res.status(403).json({ error: 'Platform admin access required' })
  }

  next()
}

module.exports = { verifyToken, requireRole, requirePlatformAdmin }

function requireRestaurantAccess(req, res, next) {
  const restaurantId = restaurantIdFromRequest(req)

  if (!restaurantId || isPlatformAdmin(req.user)) {
    return next()
  }

  if (!req.user?.restaurantId) {
    return res.status(403).json({
      success: false,

    })

    return res.status(403).json({
      success: false,
      error: 'You do not have access to this restaurant',
    })
  }

  next()

async function requireActiveSubscription(req, res, next) {
  try {
    const restaurantId = restaurantIdFromRequest(req)

    if (!restaurantId) {
      return res.status(400).json({
        success: false,
        error: 'restaurantId is required',
      })
    }

    const restaurant = await getRestaurant(restaurantId)

    if (!restaurant) {
      return res.status(404).json({
        success: false,
        error: 'Restaurant not found',
      })
    }

    req.restaurant = restaurant

    if (!restaurant.subscription.isServiceEnabled) {
      return res.status(402).json({
        success: false,
        error: 'Subscription expired',
        subscription: restaurant.subscription,
      })
    }

    next()
  } catch (error) {
    res.status(500).json({
      success: false,
      error: error.message,
    })
  }
}

const { z } = require('zod')

/**
 * Express middleware factory: validates req.body against a Zod schema.
 * Usage: router.post('/path', validate(mySchema), handler)
 */
function validate(schema) {
  return (req, res, next) => {
    try {
      req.body = schema.parse(req.body)
      next()
    } catch (err) {
      if (err instanceof z.ZodError) {
        const messages = err.errors
          ? err.errors.map((e) => `${e.path.join('.')}: ${e.message}`)
          : ['Validation failed']
        return res.status(400).json({
          success: false,
          error: 'Validation failed',
          details: messages,
        })
      }
      next(err)
    }
  }
}

  limits: { fileSize: 5 * 1024 * 1024 },
  fileFilter: (req, file, cb) => {

    }

  '/upload',
  verifyToken,
  requireRole('admin'),
  upload.single('image'),
  async (req, res) => {

      if (!req.file) {
        return res.status(400).json({ success: false, error: 'No image file provided' })
      }

      const restaurantId = req.body.restaurantId || req.user.restaurantId
      if (!restaurantId) {
        return res.status(400).json({ success: false, error: 'restaurantId is required' })
      }

      // Upload to Cloudinary
      const result = await new Promise((resolve, reject) => {
        const stream = cloudinary.uploader.upload_stream(
          {
            folder: `digital-menu/${restaurantId}`,
            resource_type: 'image',
            transformation: [
              { width: 800, height: 600, crop: 'limit' },
              { quality: 'auto', fetch_format: 'auto' }
            ]
          },
          (error, result) => {
            if (error) reject(error)
            else resolve(result)
          }
        )
        stream.end(req.file.buffer)
      })

      res.json({
        success: true,
        url: result.secure_url,
        publicId: result.public_id,
      })

const deriveOrderStatus = (items = []) => {
  const statuses = items.map((item) => item.status || 'new')
  if (statuses.length === 0) return 'new'
  if (statuses.every((status) => status === 'served')) return 'served'
  if (statuses.every((status) => status === 'ready' || status === 'served')) return 'ready'
  if (statuses.some((status) => status === 'preparing' || status === 'ready' || status === 'served')) {
    return 'preparing'

  return 'new'
}

const toNumber = (value, fallback = 0) => {
  const number = Number(value)

const itemSignature = (items = []) =>
  normalizeItems(items)
    .map((item) => ({
      id: item.id,
      price: toNumber(item.price),
      quantity: toNumber(item.quantity),
      variant: item.variant || '',
      specialInstructions: (item.specialInstructions || '').trim(),

const assertNotDuplicateOrder = (orders, orderData) => {
  const now = Date.now()
  const incomingNote = (orderData.specialInstructions || '').trim()

  const duplicate = orders.find((order) => {
    const timestamp = new Date(order.updatedAt || order.createdAt).getTime()
    if (!timestamp || now - timestamp > DUPLICATE_WINDOW_MS) return false
    const existingItems = order.newItems?.length ? order.newItems : order.items
    const existingNote = (order.specialInstructions || '').trim()
    return incomingNote === existingNote && sameOrderPayload(existingItems, orderData.items || [])
  })

  if (duplicate) {
    const error = new Error('Duplicate order blocked. Please wait a few seconds before ordering the same items again.')
    error.statusCode = 409
    throw error
  }

const buildSettlement = (subtotal, options = {}) => {
  const discountAmount = Math.min(Math.max(toNumber(options.discountAmount), 0), subtotal)
  const serviceChargeAmount = Math.max(toNumber(options.serviceChargeAmount), 0)
  const gstPercent = Math.max(toNumber(options.gstPercent), 0)
  const taxableAmount = Math.max(subtotal - discountAmount + serviceChargeAmount, 0)
  const gstAmount = roundMoney((taxableAmount * gstPercent) / 100)
  const payableAmount = roundMoney(taxableAmount + gstAmount)

  return {
    subtotal: roundMoney(subtotal),
    discountAmount: roundMoney(discountAmount),
    serviceChargeAmount: roundMoney(serviceChargeAmount),
    gstPercent: roundMoney(gstPercent),
    gstAmount,
    payableAmount,
    paymentMode: options.paymentMode || 'cash',
    paymentReference: options.paymentReference || '',
    notes: options.notes || '',
  }
}

const updateOrder = async (restaurantId, orderId, updates) => {

  const now = new Date().toISOString()

  if (updates.status === 'preparing' || updates.status === 'ready' || updates.status === 'served') {

    const existing = doc.exists ? doc.data() : {}
    data.items = normalizeItems(existing.items, existing.status || 'new').map((item) =>

    )

      item.status === 'served' ? item : { ...item, status: updates.status }
    )
  }

  if (updates.status === 'ready') data.completedAt = now
  if (updates.status === 'served') data.servedAt = now
  if (updates.status === 'closed') data.closedAt = now

      if (doc.exists) ordersToClose.push({ id: doc.id, ...doc.data() })
    }
  } else {
    const snap = await orderCol(restaurantId)
      .where('tableNumber', '==', String(tableNumber))
      .where('status', '!=', 'closed')
      .get()
    ordersToClose = snap.docs.map(d => ({ id: d.id, ...d.data() }))
  }

  const settlement = buildSettlement(totalAmount, settlementOptions)

  ordersToClose.forEach(order => {
    const ref = orderCol(restaurantId).doc(order.id)
    const ratio = totalAmount > 0 ? order.totalAmount / totalAmount : 0
    const settledAmount = roundMoney(settlement.payableAmount * ratio)
    batch.update(ref, {
      status: 'closed',
      closedAt: now,
      updatedAt: now,
      settledAmount,
      settlement: {
        ...settlement,
        orderSubtotal: roundMoney(order.totalAmount),
        settledAmount,
      },
    })
  })

  await batch.commit()

  return {
    restaurantId,
    tableNumber,
    orderIds: ordersToClose.map(o => o.id),
    totalAmount,
    settledAmount: settlement.payableAmount,
    settlement,
    closedAt: now,
  }

  const totalRevenue = todayClosed.reduce(
    (sum, o) => sum + (o.settledAmount ?? o.totalAmount ?? 0), 0
  )

  const paymentModes = {}
  todayClosed.forEach((o) => {
    const mode = o.settlement?.paymentMode || 'cash'
    if (!paymentModes[mode]) paymentModes[mode] = { count: 0, total: 0 }
    paymentModes[mode].count += 1
    paymentModes[mode].total += o.settledAmount ?? o.totalAmount ?? 0
  })

  const itemCounts = {}
  todayOrders.forEach((o) => {
    ;(o.items || []).forEach((item) => {
      const key = item.name || item.id
      if (!itemCounts[key]) itemCounts[key] = { name: key, quantity: 0, revenue: 0 }
      itemCounts[key].quantity += item.quantity || 1
      itemCounts[key].revenue += (item.price || 0) * (item.quantity || 1)
    })
  })
  const topItems = Object.values(itemCounts)
    .sort((a, b) => b.quantity - a.quantity)
    .slice(0, 5)

  const hourlyBreakdown = Array.from({ length: 24 }, (_, i) => ({ hour: i, orders: 0 }))
  todayOrders.forEach((o) => {
    const h = new Date(o.createdAt).getHours()
    if (h >= 0 && h < 24) hourlyBreakdown[h].orders += 1
  })

  return {
    date: new Date().toISOString().split('T')[0],
    totalOrders: todayClosed.length,
    totalRevenue: roundMoney(totalRevenue),
    averageOrderValue: todayClosed.length > 0 ? roundMoney(totalRevenue / todayClosed.length) : 0,
    paymentModeBreakdown: paymentModes,
    topItems,
    hourlyBreakdown,
  }
}

let orders = []

function isToday(dateValue) {
  if (!dateValue) return false

  const date = new Date(dateValue)
  const today = new Date()

  return (
    date.getFullYear() === today.getFullYear() &&
    date.getMonth() === today.getMonth() &&
    date.getDate() === today.getDate()
  )
}

function createOrder(orderData) {
  const totalAmount = orderData.totalAmount ??
    orderData.items.reduce((sum, item) => sum + item.price * item.quantity, 0)

  const newOrder = {
    id: orderData.id || Date.now().toString(),
    ...orderData,
    totalAmount,
    status: orderData.status || 'new',
    createdAt: orderData.createdAt || new Date().toISOString(),
    updatedAt: new Date().toISOString(),
  }

  orders.push(newOrder)
  return newOrder
}

function upsertOrder(orderData) {
  const index = orders.findIndex((order) => order.id === orderData.id)

  if (index === -1) {
    return createOrder(orderData)
  }

  orders[index] = {
    ...orders[index],
    ...orderData,
    updatedAt: new Date().toISOString(),
  }

  return orders[index]
}

function getRestaurantOrders(restaurantId) {
  return orders.filter(
    (order) => order.restaurantId === restaurantId && order.status !== 'closed'
  )
}

function updateOrder(orderId, updates) {
  const orderIndex = orders.findIndex((order) => order.id === orderId)

  if (orderIndex === -1) {
    return null
  }

  orders[orderIndex] = {
    ...orders[orderIndex],
    ...updates,
    updatedAt: new Date().toISOString(),
  }

  if (updates.status === 'ready' && !orders[orderIndex].completedAt) {
    orders[orderIndex].completedAt = new Date().toISOString()
  }

  return orders[orderIndex]
}

function updateOrderStatus(orderData, status) {
  return updateOrder(orderData.id, {
    ...orderData,
    status,
  }) || upsertOrder({ ...orderData, status })
}

function closeBill(orderData) {
  const now = new Date().toISOString()
  const orderIds = orderData.orderIds || (orderData.id ? [orderData.id] : [])
  const matchingOrders = orders.filter((order) => {
    if (order.restaurantId !== orderData.restaurantId) return false
    if (orderIds.length > 0) return orderIds.includes(order.id)
    return String(order.tableNumber) === String(orderData.tableNumber) && order.status !== 'closed'
  })

  matchingOrders.forEach((order) => {
    order.status = 'closed'
    order.closedAt = now
    order.updatedAt = now
  })

  return {
    ...orderData,
    orderIds: matchingOrders.map((order) => order.id),
    totalAmount: matchingOrders.reduce((sum, order) => sum + order.totalAmount, 0),
    closedAt: now,
  }
}

  )
  const todayCompletedOrders = restaurantOrders.filter((order) => isToday(order.completedAt))

  return {
    dailyRevenue: todayClosedOrders.reduce((sum, order) => sum + order.totalAmount, 0),
    completedOrdersToday: todayCompletedOrders.length,
  }
}

const TRIAL_DAYS = 15
const BILLING_DAYS = 30
const PLAN_AMOUNT = 2000

function addDays(date, days) {
  const next = new Date(date)
  next.setDate(next.getDate() + days)
  return next
}

function toIso(date) {
  return date.toISOString()
}

function getDefaultSubscription(now = new Date()) {
  return {
    subscriptionStatus: 'trial',
    trialStartedAt: toIso(now),
    trialEndsAt: toIso(addDays(now, TRIAL_DAYS)),
    currentPeriodStart: null,
    currentPeriodEnd: null,
    planAmount: PLAN_AMOUNT,
    lastPaidAt: null,
    suspendedAt: null,
  }
}

function getSubscriptionState(restaurant, now = new Date()) {
  const status = restaurant?.subscriptionStatus || 'trial'
  const createdAt = restaurant?.createdAt ? new Date(restaurant.createdAt) : now
  const lastPaidAt = restaurant?.lastPaidAt ? new Date(restaurant.lastPaidAt) : createdAt
  const trialEndsAt = restaurant?.trialEndsAt
    ? new Date(restaurant.trialEndsAt)
    : addDays(createdAt, TRIAL_DAYS)
  const currentPeriodEnd = restaurant?.currentPeriodEnd
    ? new Date(restaurant.currentPeriodEnd)
    : addDays(lastPaidAt, BILLING_DAYS)

  const isTrialValid = status === 'trial' && trialEndsAt && trialEndsAt >= now
  const isPaidValid = status === 'active' && currentPeriodEnd && currentPeriodEnd >= now
  const isSuspended = status === 'suspended'
  const isServiceEnabled = !isSuspended && (isTrialValid || isPaidValid)

  let effectiveStatus = status
  if (!isServiceEnabled && status !== 'suspended') {
    effectiveStatus = 'expired'
  }

  return {
    effectiveStatus,
    isServiceEnabled,
    planAmount: restaurant?.planAmount || PLAN_AMOUNT,
    trialEndsAt: toIso(trialEndsAt),
    currentPeriodEnd: toIso(currentPeriodEnd),
    daysRemaining: getDaysRemaining(
      status === 'trial' ? trialEndsAt : currentPeriodEnd,
      now
    ),
  }
}

function getDaysRemaining(endDate, now = new Date()) {
  if (!endDate) return 0
  const diff = endDate.getTime() - now.getTime()
  return Math.max(0, Math.ceil(diff / (1000 * 60 * 60 * 24)))
}

function buildSubscriptionUpdate(action, options = {}) {
  const now = new Date()
  const days = Number(options.days) || BILLING_DAYS

  if (action === 'markPaid') {
    return {
      subscriptionStatus: 'active',
      currentPeriodStart: toIso(now),
      currentPeriodEnd: toIso(addDays(now, days)),
      lastPaidAt: toIso(now),
      planAmount: PLAN_AMOUNT,
      suspendedAt: null,
    }
  }

  if (action === 'extendTrial') {
    const trialDays = Number(options.days) || TRIAL_DAYS
    return {
      subscriptionStatus: 'trial',
      trialEndsAt: toIso(addDays(now, trialDays)),
      planAmount: PLAN_AMOUNT,
      suspendedAt: null,
    }
  }

  if (action === 'suspend') {
    return {
      subscriptionStatus: 'suspended',
      suspendedAt: toIso(now),
    }
  }

  if (action === 'reactivate') {
    return {
      subscriptionStatus: 'active',
      currentPeriodStart: toIso(now),
      currentPeriodEnd: toIso(addDays(now, days)),
      lastPaidAt: toIso(now),
      planAmount: PLAN_AMOUNT,
      suspendedAt: null,
    }
  }

  throw new Error('Invalid subscription action')
}

          orderId: data.orderId,
          tableId: data.tableId,
          tableNumber: data.tableNumber,
          message: data.message,
          createdAt: new Date().toISOString(),
        })
      } catch (err) {
        console.error('chef_message error:', err)
      }
    })

    socket.on('close_bill', async (data) => {
      try {
        if (!(await isServiceEnabled(data.restaurantId))) {
          return socket.emit('subscription_blocked', { restaurantId: data.restaurantId })
        }
        const result = await closeBill(data.restaurantId, data.tableNumber, data.orderIds, data.settlement)
        io.to(data.restaurantId).emit('bill_closed', result)
        await emitStats(io, data.restaurantId)
      } catch (err) {
        console.error('close_bill error:', err)
      }
    })

    socket.on('call_waiter', async (data) => {
      try {
        if (!data.restaurantId || !data.tableId) return
        if (!(await isServiceEnabled(data.restaurantId))) {
          return socket.emit('subscription_blocked', { restaurantId: data.restaurantId })
        }
        io.to(data.restaurantId).emit('waiter_called', {
          restaurantId: data.restaurantId,
          tableId: data.tableId,
          tableNumber: data.tableNumber ?? data.tableId,
          calledAt: new Date().toISOString(),
        })
      } catch (err) {
        console.error('call_waiter error:', err)
      }
    })

    socket.on('disconnect', () => {
      console.log('Disconnected:', socket.id)
    })
  })
}

module.exports = { initOrderSocket }
